1 Data Construction

The data is derived from PASSIM Core [@schmidt_computational_2023]. It creates a unimode network of sermons connected with sermons by their co-occurrance in a manuscript. It is also a directional network, because the order in which the manuscripts occur is also relevant. @french_research Highlighted reoccuring blocks of sermons at the end of manuscripts. Furthermore, the number of edges is significantly reduced from c. 800.000 to c. 26.000 which is computationally benefitial when working with stocastic models.

The manuscripts function as observations to observe relationships between sermons. The observations are samples and its not conclusive nor exhaustive. Nevertheless, we’re interested in the mechanisms behind the co-occurrance of sermons and extensive sampling suffices to this end.

The sermons dont have agency by themselves by compilers chose to copy sermons in a certain order and selected them deliberatly, given the availablity of source material, in order words, the script picked sermons to copy from selection of sermons they had access to.

import pandas as pd
# create edgelist per manuscript undirected
def undirected_se_se (se_list:list, ms:str, century:int|str = None) -> list[list]:
    sub_se_se_l = []
    for se1 in se_list:
        for se2 in se_list:
            if se1 > se2:
                sub_se_se_l.append((se1, se2, ms, century, 'undirected'))
    return sub_se_se_l

# create edgelist per manuscript directed
def directed_se_se (se_list:list, ms:str, century:int|str = None) -> list[list]:
    sub_se_se_l = []

    # iterate over length sermon list and connect a sermon to its preceding sermon
    for i in range(len(se_list)-1):
        se1 = se_list[i]
        se2 = se_list[i+1]
        sub_se_se_l.append((se1, se2, ms, century, 'directed'))
    return sub_se_se_l

# create se-se edgelist with century, directed, manuscript attributes
def se_se_nx_df (ms_se_df: pd.DataFrame, directed:bool=False)-> pd.DataFrame:
    se_se_l = []
    for century, sub_df in ms_se_df.groupby('century'):
        for ms, ses_str in zip(sub_df['shelfmark'], sub_df['certain']):
            if isinstance(ses_str, str):
                # split the sermon string like "se-1, se-3 etc." into list
                ses = [se for se in ses_str.split(', ') if se and isinstance(se, str)]

                # make unique and without order for undirected network
                ses = ses if directed else list(set(ses))

                # process into list of edge per manuscript
                sub_se_se = directed_se_se(ses, ms, century) if directed else undirected_se_se(ses, ms, century)

                # add list of edges per ms into big edge list
                [se_se_l.append(se_se) for se_se in sub_se_se]

    se_se_df = pd.DataFrame(se_se_l, columns = ['Source', 'Target', 'manuscript', 'century', 'Directed'])
    se_se_df['Id'] = list(range(len(se_se_l)))

    return se_se_df

# create nodes of sermons with HC presence as attribute
def se_nodes_nx_df (se_se_df: pd.DataFrame, colls_d: dict) -> pd.DataFrame:
    ses = set(list(se_se_df['Source']) + list(se_se_df['Target']))

    nodes_l = []

    # Add one if the sermon is present in the list of a collection
    for se in ses:
        sub_l = [se]
        for col, col_ses in colls_d.items():
            sub_l.append(1) if se in col_ses else sub_l.append(0)
        nodes_l.append(sub_l)

    return pd.DataFrame(nodes_l, columns=['Id']+list(colls_d.keys()))

# make dict exp: {coll1 : [se1, se2, etc.], coll2 : etc.}
def collection_dictionary_list (coll_df: pd.DataFrame) -> dict:
    coll_d = coll_df.to_dict('series')

    coll_dict = {k: v.to_list() for k, v in coll_d.items()}
    coll_d_clean = {k: [e for e in v if isinstance(e, str)] for k, v in coll_dict.items()}

    return coll_d_clean
